% Man kann relativ einfach eigene Befehle in LaTeX definieren, die dann z.B. als
% simple Abkürzungen...
\newcommand{\LG}{LaTeX-Grundlagen }

% ...oder zur semantischen Auszeichnung des Textes dienen. Dadurch kann man dann
% später durch eine Änderung an dieser einen Stelle alle "Rollen" anstatt fett
% kursiv darstellen.
% in eckigen Klammern: wie viele Parameter?
% dahinter in geschweiften Klammern: was soll ausgegeben werden?
\newcommand{\Name}[1]{\textbf{#1}}

\newcommand{\Begriff}[1]{\textit{#1}}

% bei code zB nützlich für klassennamen oä, damit das einheitlich formatiert wird.

% Hier Befehl für Querverweis. 1 ist das Argument, also das Label, wo hin referenziert
% werden soll. mit #1 wird hier das Argument eingefügt.
\newcommand{\Tabelle}[1]{Tabelle \ref{tab:#1}: \nameref{tab:#1} auf Seite \pageref{tab:#1}}
\newcommand{\Kapitel}[1]{Kapitel \ref{sec:#1}: \nameref{sec:#1} auf Seite \pageref{sec:#1}}
